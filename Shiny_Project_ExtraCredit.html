The server

server <- function(input, output, session) {

  # --- Dynamic element selector based on dataset choice ----------------------
  output$element_selector <- renderUI({
    if (input$dataset == "d1") {
      selectInput("element", "Select element:", choices = d1_elements, selected = "P")
    } else {
      selectInput("element", "Select element:", choices = d3_elements, selected = "P")
    }
  })

  # --- Reactive: filtered dataset for main plots ----------------------------
  active_data <- reactive({
    req(input$soil_filter)
    if (input$dataset == "d1") {
      d1 |> filter(Depth_cm >= input$depth_range[1],
                   Depth_cm <= input$depth_range[2])
    } else {
      d3 |> filter(Soil_type %in% input$soil_filter,
                   Depth_cm  >= input$depth_range[1],
                   Depth_cm  <= input$depth_range[2])
    }
  })

  # --- TAB 1: Depth Profile Plot --------------------------------------------
  output$depth_plot <- renderPlot({
    req(input$element)
    df <- active_data()

    if (input$dataset == "d1") {
      # d1 has no soil type column — show element distribution as histogram
      ggplot(df, aes(x = .data[[input$element]])) +
        geom_histogram(fill = "#2166ac", color = "white", bins = 30) +
        labs(
          title    = paste0("Distribution of ", input$element,
                            " — Extractable Elements (d1)"),
          subtitle = paste0("All samples collected at 20 cm depth | n = ", nrow(df)),
          x        = input$element,
          y        = "Count"
        ) +
        theme_ade +
        if (input$log_y) scale_x_log10()

    } else {
      # d3: depth profile by soil type
      df_sel <- df |>
        filter(Soil_type %in% input$soil_filter) |>
        select(Depth_cm, Soil_type, val = all_of(input$element)) |>
        filter(!is.na(val))

      # Summary for ribbon
      df_sum <- df_sel |>
        group_by(Depth_cm, Soil_type) |>
        summarise(
          mean_val = mean(val, na.rm = TRUE),
          se       = sd(val, na.rm = TRUE) / sqrt(n()),
          .groups  = "drop"
        ) |>
        mutate(
          lo = mean_val - 1.96 * se,
          hi = mean_val + 1.96 * se
        )

      p <- ggplot(df_sum, aes(x = mean_val, y = Depth_cm, color = Soil_type, fill = Soil_type)) +
        scale_y_reverse(breaks = seq(0, 200, 20)) +
        scale_color_manual(values = pal, name = "Soil type") +
        scale_fill_manual(values  = pal, name = "Soil type")

      if (input$show_ci) {
        p <- p + geom_ribbon(aes(xmin = lo, xmax = hi), alpha = 0.15, color = NA)
      }

      if (input$show_pts) {
        p <- p + geom_point(data = df_sel,
                            aes(x = val, y = Depth_cm, color = Soil_type),
                            alpha = 0.25, size = 1.2)
      }

      p <- p +
        geom_path(linewidth = 1.2) +
        geom_point(size = 2.5) +
        labs(
          title    = paste0(input$element, " by Depth — ADE vs Ultisol"),
          subtitle = "Total elements, maximum contrast profiles (d3)",
          x        = paste0(input$element, " (mg/kg)"),
          y        = "Depth (cm)"
        ) +
        theme_ade

      if (input$log_y) p <- p + scale_x_log10()
      p
    }
  })

  # --- TAB 2: Boxplot + t-test ----------------------------------------------
  output$box_plot <- renderPlot({
    req(input$element, length(input$soil_filter) > 0)

    if (input$dataset == "d1") {
      df <- d1 |>
        select(val = all_of(input$element)) |>
        mutate(Soil_type = "All (no type in d1)") |>
        filter(!is.na(val))
      ggplot(df, aes(x = Soil_type, y = val, fill = Soil_type)) +
        geom_boxplot(alpha = 0.7, outlier.shape = 21) +
        geom_jitter(width = 0.15, alpha = 0.3, size = 1.5) +
        labs(title = paste0(input$element, " — No soil type grouping in d1"),
             subtitle = "Switch to d3 for ADE vs Ultisol comparison",
             y = input$element, x = "") +
        theme_ade + theme(legend.position = "none")

    } else {
      df <- d3 |>
        filter(Soil_type %in% input$soil_filter) |>
        select(Soil_type, val = all_of(input$element)) |>
        filter(!is.na(val))

      p <- ggplot(df, aes(x = Soil_type, y = val, fill = Soil_type)) +
        geom_boxplot(alpha = 0.75, outlier.shape = 21, width = 0.5) +
        geom_jitter(aes(color = Soil_type), width = 0.15, alpha = 0.4, size = 1.8) +
        scale_fill_manual(values  = pal) +
        scale_color_manual(values = pal) +
        labs(
          title    = paste0(input$element, " — ADE vs Ultisol"),
          subtitle = "Total elements, maximum contrast profiles (d3)",
          x        = "Soil type",
          y        = paste0(input$element, " (mg/kg)")
        ) +
        theme_ade + theme(legend.position = "none")

      if (input$log_y) p <- p + scale_y_log10()
      p
    }
  })

  output$ttest_table <- renderTable({
    req(input$dataset == "d3", length(input$soil_filter) == 2)
    df <- d3 |>
      filter(Soil_type %in% c("ADE", "Ultisol")) |>
      select(Soil_type, val = all_of(input$element)) |>
      filter(!is.na(val))

    tt <- tryCatch(
      t.test(val ~ Soil_type, data = df),
      error = function(e) NULL
    )
    if (is.null(tt)) return(data.frame(Note = "Could not compute t-test (insufficient data)"))

    sig <- ifelse(tt$p.value < 0.001, "***",
           ifelse(tt$p.value < 0.01,  "**",
           ifelse(tt$p.value < 0.05,  "*", "ns")))

    data.frame(
      Element        = input$element,
      `Mean ADE`     = round(tt$estimate[2], 3),
      `Mean Ultisol` = round(tt$estimate[1], 3),
      `t statistic`  = round(tt$statistic, 3),
      df             = round(tt$parameter, 1),
      `p-value`      = formatC(tt$p.value, format = "e", digits = 3),
      Significance   = sig,
      check.names    = FALSE
    )
  }, striped = TRUE, bordered = TRUE, spacing = "s", align = "c", width = "100%")

  # --- TAB 3: Bivariate Regression ------------------------------------------
  output$reg_plot <- renderPlot({
    req(input$reg_x, input$reg_y)
    df <- d3 |>
      select(Soil_type, x = all_of(input$reg_x), y = all_of(input$reg_y)) |>
      filter(!is.na(x), !is.na(y))

    if (input$reg_log) {
      df <- df |> filter(x > 0, y > 0) |>
        mutate(x = log10(x), y = log10(y))
    }

    if (input$reg_color) {
      p <- ggplot(df, aes(x = x, y = y, color = Soil_type)) +
        scale_color_manual(values = pal, name = "Soil type")
    } else {
      p <- ggplot(df, aes(x = x, y = y))
    }

    p +
      geom_point(alpha = 0.6, size = 2) +
      geom_smooth(method = "lm", se = TRUE, linewidth = 1) +
      labs(
        title    = paste0(if (input$reg_log) "log10(" else "",
                          input$reg_y,
                          if (input$reg_log) ")" else "",
                          " ~ ",
                          if (input$reg_log) "log10(" else "",
                          input$reg_x,
                          if (input$reg_log) ")" else ""),
        subtitle = "Total elements — maximum contrast profiles (d3)",
        x        = paste0(if (input$reg_log) "log10(" else "", input$reg_x,
                          if (input$reg_log) ")" else ""),
        y        = paste0(if (input$reg_log) "log10(" else "", input$reg_y,
                          if (input$reg_log) ")" else "")
      ) +
      theme_ade
  })

  output$reg_model_table <- renderTable({
    req(input$reg_x, input$reg_y)
    df <- d3 |>
      select(x = all_of(input$reg_x), y = all_of(input$reg_y)) |>
      filter(!is.na(x), !is.na(y))
    if (input$reg_log) df <- df |> filter(x > 0, y > 0) |>
      mutate(x = log10(x), y = log10(y))

    m <- lm(y ~ x, data = df)
    tidy(m) |>
      rename(Term = term, Estimate = estimate,
             `Std. Error` = std.error,
             `t-value` = statistic, `p-value` = p.value) |>
      mutate(across(where(is.numeric), ~ round(.x, 4)))
  }, striped = TRUE, bordered = TRUE, spacing = "s", align = "c", width = "100%")

  output$reg_glance_table <- renderTable({
    req(input$reg_x, input$reg_y)
    df <- d3 |>
      select(x = all_of(input$reg_x), y = all_of(input$reg_y)) |>
      filter(!is.na(x), !is.na(y))
    if (input$reg_log) df <- df |> filter(x > 0, y > 0) |>
      mutate(x = log10(x), y = log10(y))

    m <- lm(y ~ x, data = df)
    glance(m) |>
      select(r.squared, adj.r.squared, sigma, statistic, p.value, nobs) |>
      rename(`R²` = r.squared, `Adj. R²` = adj.r.squared,
             `Residual SE` = sigma, `F-statistic` = statistic,
             `p-value` = p.value, `n` = nobs) |>
      mutate(across(where(is.numeric), ~ round(.x, 4)))
  }, striped = TRUE, bordered = TRUE, spacing = "s", align = "c", width = "100%")

  # --- TAB 4: Correlation Matrix --------------------------------------------
  output$corr_plot <- renderPlot({
    if (input$corr_dataset == "d3") {
      df <- d3
      if (input$corr_soil != "All") df <- df |> filter(Soil_type == input$corr_soil)
      vars <- c("P", "Ca", "Mg", "K", "Fe", "Al", "Ti", "Mn",
                "Sr", "Ba", "Zn", "Cu", "Pb", "V", "Cr", "Rb")
      df_num <- df |> select(all_of(vars)) |> na.omit()
    } else {
      df <- d1
      vars <- c("P", "K", "Na", "Ca", "Mg", "Al", "Fe", "Zn", "Mn", "Cu",
                "pH_H2O", "pH_K2O")
      df_num <- df |> select(all_of(vars)) |> na.omit()
    }

    cor_mat <- cor(df_num, use = "complete.obs")
    title_txt <- paste0(
      "Pearson Correlations — ",
      if (input$corr_dataset == "d3") "Total Elements (d3)" else "Extractable Elements (d1)",
      if (input$corr_soil != "All") paste0(" | ", input$corr_soil, " only") else " | All soils"
    )

    corrplot(
      cor_mat,
      method      = "color",
      type        = "upper",
      tl.col      = "black",
      tl.srt      = 45,
      tl.cex      = 0.85,
      addCoef.col = "black",
      number.cex  = 0.60,
      col         = colorRampPalette(c("#d6604d", "white", "#2166ac"))(200),
      title       = title_txt,
      mar         = c(0, 0, 2, 0)
    )
  })

  # --- TAB 5: Spatial Map ---------------------------------------------------
  output$map_plot <- renderPlot({
    req(input$map_element)
    df_map <- d1 |>
      select(X, Y, val = all_of(input$map_element)) |>
      filter(!is.na(val))

    fill_vals <- if (input$map_log) log10(df_map$val + 1) else df_map$val
    df_map$fill_val <- fill_vals
    label_txt <- if (input$map_log) paste0("log10(", input$map_element, "+1)")
                 else input$map_element

    ggplot(df_map, aes(x = X, y = Y, color = fill_val)) +
      geom_point(size = 3.5, alpha = 0.85) +
      scale_color_viridis_c(
        option = "plasma",
        name   = label_txt,
        direction = -1
      ) +
      coord_equal() +
      labs(
        title    = paste0("Spatial Distribution of ", input$map_element),
        subtitle = "Extractable elements reference profiles (d1) | Central Amazon, Brazil",
        x        = "UTM Easting (m)",
        y        = "UTM Northing (m)"
      ) +
      theme_ade +
      theme(legend.position = "right")
  })

  # --- TAB 6: Data Table ----------------------------------------------------
  output$data_table <- DT::renderDataTable({
    if (input$table_ds == "d1") {
      d1 |> mutate(across(where(is.numeric), ~ round(.x, 3)))
    } else {
      d3 |> mutate(across(where(is.numeric), ~ round(.x, 3)))
    }
  },
  options  = list(
    pageLength = 10,
    scrollX    = TRUE,
    lengthMenu = list(c(10, 25, 50, -1), c("10", "25", "50", "All"))
  ),
  rownames = FALSE
  )
}
